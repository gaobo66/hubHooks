function debounce(func: Function, delay: number) {
    let timer: NodeJS.Timeout;
    return function (...args: any[]) {
        clearTimeout(timer);
        timer = setTimeout(() => {
            func(...args);
        }, delay);
    };
}

    function throttleTrailing<T extends (...args: any[]) => any>(
  func: T,
  delay: number 
): ThrottleFunction<T> {
  let timer: NodeJS.Timeout | null = null; //
  let lastArgs: Parameters<T> | null = null;//最新参数
  let lastCall = 0; // 

  return function throttled(...args: Parameters<T>) {
    const now = Date.now();
   // 计算距离下次可以调用目标函数的剩余时间   
    const remaining = delay - (now - lastCall);

    if (remaining <= 0 || remaining > delay) { // 原有条件
      if (lastArgs) {
        func(...lastArgs);
        lastArgs = null;
      }
      lastCall = now;
      // 不需要清除定时器，因为此时定时器还没有被设置，或者已经被清除
    } else {
      lastArgs = args;
      // 设置定时器在剩余时间后调用目标函数
      if (!timer) {
        timer = setTimeout(() => {
          if (lastArgs) {
            func(...lastArgs);
            lastArgs = null;
          }
          lastCall = Date.now();
          timer = null;
        }, remaining);
      }
    }
  };
}

/**
 * @description arrayToTree 数组转树
 * @param {*} items 数组
 * @param {*} id 数组项的id
 * @param {*} parentId 数组项的parentId
 * @param {*} children 数组项的children
 * @returns 数组转树
 */

export function arrayToTree(items, id = 'id', parentId = 'parentId', children = 'children') {
    const itemMap = new Map();
    const tree = [];

    // 构建映射表
    items.forEach(item => {
        itemMap.set(item[id], {...item, [children]: []});
    });

    // 构建树结构
    items.forEach(item => {
        if (item[parentId]) {
            itemMap.get(item[parentId])[children].push(itemMap.get(item[id]));
        } else {
            tree.push(itemMap.get(item[id]));
        }
    });

    return tree;
}

// v2.0
export function arrayToTreeV2(items, id = 'id', parentId = 'parentId', children = 'children') {
    const itemMap = new Map();

    return items.reduce((tree, item) => {
        // 将当前项添加到映射表中
        itemMap.set(item[id], {...item, [children]: []});

        // 如果存在 parentId，则将当前项添加到父节点的 children 中
        if (item[parentId] && itemMap.get(item[parentId])) {
            itemMap.get(item[parentId])[children].push(itemMap.get(item[id]));
        } else {
            // 否则作为根节点添加到树中
            tree.push(itemMap.get(item[id]));
        }

        return tree;
    }, []);
}

/**
 * @description treeToArray 树转数组
 * @param {*} tree 树
 * @param {*} children 数组项的children
 * @param {*} children 子节点字段名，默认 'children'
 * @returns 转换后的数组，包含父子关系信息
 */
export function treeToArray(tree, children = 'children') {
    const result = [];

    function traverse(node, parentId = null) {
        if (!node) return;

        // 复制节点并添加 parentId
        const {[children]: childNodes, ...rest} = node;
        result.push({...rest, parentId});

        // 递归处理子节点
        if (childNodes && childNodes.length > 0) {
            childNodes.forEach(child => traverse(child, node.id));
        }
    }

    tree.forEach(traverse);
    return result;
}

/**
 * @description findNode 查找树节点
 * @param {*} tree 树
 * @param {*} id 数组项的id
 * @returns 查找树节点
 */
unction findTreeNode(tree, id) {
   let queue = []; // 定义队列
   queue.push(tree); // 将根节点入队
   let head = 0; // 队头索引
   while (queue.length) {
      // 当队列不为空时
      const node = queue[head++]; // 取出队头节点
      if (node.id === id) return node; // 如果节点的 id 等于目标 id，返回该 节点
      if(node?.children?.length){ // 如果节点有子节点
         queue.push(...node.children); // 将子节点入队
      }
      if (head >= 1000) {
         queue = queue.slice(0,head); // 如果队列长度超过 1000，截断队列
         head = 0; // 重置队头索引
      }

   }
   return undefined; // 如果队列为空，返回 undefined
}

function findTreeNodeRecursive(tree, id) {
   if (tree.id === id) return tree; // 如果当前节点的 id 等于目标 id，返回该节点
   for (const child of tree.children) { // 遍历子节点
      const result = findTreeNodeRecursive(child, id); // 递归查找子节点
      if (result) return result; // 如果找到，返回结果
   }
   return undefined; // 如果没有找到，返回 undefined
}

/**
 * @description 实现一个控制并发请求的函数-数组的并发控制
 * @param {Array<Function>} requests 异步请求函数数组
 * @param {number} maxConcurrency 最大并发数，默认为 5
 * @returns {Promise<Array>} 所有请求的结果数组
 */

function concurrentRequests(requests, maxConcurrency = 5) {
    return new Promise((resolve, reject) => {
        const results = new Array(requests.length); // 存储所有请求的结果
        let runningCount = 0; // 当前处理请求数
        let currentIndex = 0; // 当前请求的索引
        function runNext() {
            // 如果所有请求都已处理完成
            if (currentIndex >= requests.length && runningCount === 0) {
                resolve(results); // 返回所有请求的结果
                return; // 结束函数
            }

            // 如果未达到最大并发数且还有未处理的请求
            while (runningCount < maxConcurrency && currentIndex < requests.length) {
                const request = requests[currentIndex]; // 获取当前请求

                runningCount++; // 当前处理请求数加1
                request()
                    .then((res) => (results[currentIndex] = res))
                    .catch((err) => (results[currentIndex] = err))
                    .finally(() => {
                        currentIndex++; // 索引加1
                        runningCount--; // 当前处理请求数减1
                        runNext(); // 继续处理下一个请求
                    });
            }
        }

        runNext(); // 开始处理请求
    });
}

/**
 * @description 实现一个控制并发请求的函数
 * @param {Function} request 异步请求函数
 * @param {number} maxConcurrency 最大并发数，默认为 5
 * @returns {Promise} 请求的 Promise
 */
function controlConcurrency(request, maxConcurrency = 5) {
    let running = 0; // 当前正在运行的请求数量
    const queue = []; // 请求队列

    return function (...args) {
        return new Promise((resolve, reject) => {
            queue.push({request, args, resolve, reject});
            runNext();
        });
    };

    function runNext() {
        // 如果当前正在运行的请求数量小于最大并发数，并且队列中有请求
        while (running < maxConcurrency && queue.length > 0) {
            const {request, args, resolve, reject} = queue.shift(); // 取出队列中的第一个请求
            running++; // 增加正在运行的请求数量

            request(...args)
                .then((result) => {
                    resolve(result); // 请求成功，返回结果
                })
                .catch((error) => {
                    reject(error); // 请求失败，返回错误
                })
                .finally(() => {
                    running--; // 减少正在运行的请求数量
                    runNext(); // 继续执行下一个请求
                });
        }
    }
}

);

function createCannelTask(fn) {
    const Noop = () => {
    }
    let cancel = Noop
    return {
        cancel: () => cancel(),
        run: (...args) => {
            return new Promise((resolve, reject) => {
                cancel()
                cancel = () => (reject = resolve = Noop)
                fn(...args).then(
                    res => resolve(res),
                    err => reject(err),
                )
            })
        }
    }
}

// 模拟一个长时间运行的异步任务
function fetchData(delay) {
    return new Promise(resolve => {

            resolve(`Data fetched after ${delay}ms`)
        }, delay)
    })

/**
 * @description 利用闭包实现函数l柯里化
 * 柯里化是一种将接受多个参数的函数转换为一系列使用单一参数的函数的技术
 * @param {*} fn 函数
 * @returns 函数
 */
export function curry(fn) {
    return function curried(...args) {
        if (args.length >= fn.length) {
            return fn.apply(this, args);
        } else {
            return function (...moreArgs) {
                return curried.apply(this, args.concat(moreArgs))
            }
        }
    };
}

function add(a, b, c) {
    return a * b * c;
}

const curriedAdd = curry(add);

/**
 *
 * @param {*} fn
 * @returns promise
 */

function createAsyncFn(fn) {
    let p = Promise.resolve();

    return function (...args) {
        const _fn = fn.bind(this, ...args);
        return (p = p.then(
            () => _fn(),
            () => _fn()
        ));
    };
}

const asyncFn = createAsyncFn(async (x) => {
    console.time(x);
    await new Promise(resolve => setTimeout(resolve, 1000));
    console.timeEnd(x);
    return x * 2;
});

// 会按顺序执行，间隔1秒输出

class LazyManImp {
    constructor(name) {
        // 任务队列：每个任务都返回 Promise，且接收上一次结果、resolve 本次结果
        this.taskQueue = [];

        // 初始任务：同步任务包装成 Promise
        this.taskQueue.push(async (lastResult) => {
            console.log(`你好,${name}`);
            return name; // Promise 会自动 resolve 这个值
        });

        // 微任务启动队列（保证所有链式方法先入队）
        Promise.resolve().then(() => this._runNext());
    }

    // 核心执行方法：用 async/await 统一处理同步/异步任务
    async _runNext(lastResult) {
        if (this.taskQueue.length === 0) return;

        // 取出队首任务并执行
        const currentTask = this.taskQueue.shift();
        try {
            // 不管任务是同步还是异步，await 都能拿到结果
            const currentResult = await currentTask(lastResult);
            // 递归执行下一个任务，传递本次结果
            this._runNext(currentResult);
        } catch (err) {
            console.error("任务执行出错：", err);
            // 容错：即使某个任务报错，也继续执行下一个
            this._runNext(lastResult);
        }
    }

    // 链式方法：吃（同步任务，自动包装成 Promise）
    eat(food) {
        this.taskQueue.push(async (lastResult) => {
            console.log(`${lastResult} 吃${food}`);
            return lastResult; // 透传结果
        });
        return this;
    }

    // 链式方法：睡觉（异步任务，返回 Promise）
    sleep(seconds) {
        this.taskQueue.push(async (lastResult) => {
            console.log(`${lastResult}等待 ${seconds} seconds`);
            // 用 Promise 封装 setTimeout，无需手动标记
            return new Promise((resolve) => {
                setTimeout(() => {
                    resolve(lastResult); // 延迟后 resolve 结果，传给下一个任务
                }, seconds * 1000);
            });
        });
        return this;
    }

    // 链式方法：优先睡觉（插队到队列头部）
    sleepFirst(seconds) {
        this.taskQueue.unshift(async (lastResult) => {
            console.log(`${lastResult} 等待${seconds} seconds!`);
            return new Promise((resolve) => {
                setTimeout(() => {
                    resolve(lastResult);
                }, seconds * 1000);
            });
        });
        return this;
    }

    // 链式方法：工作（生成新结果，同步）
    work(job) {
        this.taskQueue.push(async (lastResult) => {
            const newResult = `${lastResult}-${job}-`;
            console.log(`${lastResult}工作是 ${job}`);
            return newResult; // 新结果传给下一个任务
        });
        return this;
    }
}

// 外层工厂函数
function lazyMan(name) {
    return new LazyManImp(name);
}


lazyMan("'Mr.G")
    .work("前端开发")
    .sleepFirst(1) // 优先睡1秒
    .sleep(1) // 优先睡1秒
    .eat("早餐")
    .sleep(2)
    .eat("午餐");

请求缓存 ​

使用场景

  1. 频繁的api请求
    1. 短时间内多次请求相同的数据
    2. 多个组件同时请求数据
  2. 多个相同的并发请求合并为一个：如在不同的组件中同时调用一个接口查询
javascript
function defaultGenKey(options) {
    return JSON.stringify(options)
}

/**
 * @description 缓存函数
 * @param fn 需要缓存的异步函数
 * @param ms 缓存时间
 * @param generatorKey 自定义生成缓存的key，默认使用JSON.stringify
 * @returns {function(...[*]): Promise<any>} 缓存异步函数
 */
function createCacheRequest(fn, ms = 0, generatorKey = defaultGenKey) {
    const map = {}
    return (...args) => {
        // 生成唯一的key，允许用户自定义
        const key = generatorKey(args)
        return (map[key] ??= fn(...args)).finally(() => {
            setTimeout(() => delete map[key], ms)
        })
    }
}

 * 模拟后端接口：根据用户ID获取用户信息
 * @param {number} userId 用户ID

 */
async function fetchUserInfo(userId) {
    // 模拟网络请求延迟（500ms）
    await new Promise(resolve => setTimeout(resolve, 500));

    // 模拟接口错误场景（方便测试catch）
    if (userId <= 0) {
        throw new Error(`无效的用户ID：${userId}，请传入正整数`);
    }

    // 打印日志：直观看到是否真的发起了API请求
    console.log(`✅ 【实际调用API】获取用户${userId}信息`);

    // 返回模拟的用户数据
    return {
        userId,
        username: `user_${userId}`,
        nickname: `昵称${userId}`,
        age: 18 + userId % 10,
        email: `user${userId}@example.com`
    };
}

// ===================== 3. 创建不同缓存策略的获取用户函数 =====================
// 策略1：缓存3秒（ms=3000 > 0 → 合并并发+缓存3秒）
const cachedFetchUser_3s = createCacheRequest(fetchUserInfo, 3000);

    console.log('========== 场景1：ms=3000 → 并发请求合并+缓存3秒 ==========');
    // 同时发起3个获取用户1的请求（并发）
    const [user1_1, user1_2, user1_3] = await Promise.all([
        cachedFetchUser_3s(1),
        cachedFetchUser_3s(1),
        cachedFetchUser_3s(1)
    ]);
    console.log('并发请求用户1的结果（3次结果一致）：', user1_1);

    // 2秒后再次请求用户1（缓存未过期，复用缓存）
    setTimeout(async () => {

    // 4秒后再次请求用户1（缓存已过期，重新调用API）
    setTimeout(async () => {
        const user1_5 = await cachedFetchUser_3s(1);
        console.log('4秒后请求用户1（缓存过期，重新调用API）：', user1_5);
    }, 4000);

    console.log('\n========== 场景2：ms=0 → 仅合并并发，不长期缓存 ==========');
    // 同时发起2个获取用户2的请求（并发合并）
    const [user2_1, user2_2] = await Promise.all([
        cachedFetchUser_0ms(2),

    ]);
    console.log('并发请求用户2的结果（2次结果一致）：', user2_1);

    // 1秒后再次请求用户2（缓存已清理，重新调用API）
    setTimeout(async () => {
        const user2_3 = await cachedFetchUser_0ms(2);
        console.log('1秒后请求用户2（无缓存，重新调用API）：', user2_3);
    }, 1000);
    
}

testUserInfoDemo();

function myNew(Constructor, ...args) {
    // 先创建一个空对象（新实例的载体）
    const obj = {}
    // 将空对象的原型指向构造函数的 prototype，保证 instanceof 等行为
    Object.setPrototypeOf(obj, Constructor.prototype)
    // 执行构造函数，把 this 绑定到新对象上
    const result = Constructor.apply(obj, args)
    // 若构造函数显式返回对象，则返回该对象，否则返回新创建的对象
    return typeof result === 'object' && result !== null ? result : obj
}

解析命令行参数 ​

javascript
let args = {}
function _booleanValue (value) {
	let  booleanValue = {
		'true': true,
		'false': false
	}[value]
	if (value === undefined) return true
	return typeof booleanValue == 'boolean' ?  booleanValue : value
}
try {
    // 解析 npm 命令传入的参数（比如 npm run dev -- --env=prod）
    JSON.parse(process.env.npm_config_argv).original.forEach((item, index) => {
        // index>1：跳过前两个默认参数（如 ["npm", "run"]）
        if (index > 1) {
            // 按等号分割参数名和值（如 "--debug=true" → ["debug=true"] → ["debug", "true"]）
            let [k, v] = item.split('=')
            // 去除参数前缀的--或-（如 --debug → debug，-env → env）
            k = k.replace(/^--?/, '')
            // 处理值并挂载到args对象
            args[k] = _booleanValue(v)
        }
    })
    // 解析原生 Node 命令传入的参数（比如 node app.js --port=3000）
    process.argv.slice(2).forEach(kv => {
        let [k, v] = kv.split('=')
        k = k.replace(/^--?/, '')
        args[k] = _booleanValue(v)
    })
} catch (e) {
    // 解析失败（比如参数格式异常、npm_config_argv格式不对）时打印提示
    console.log('解析npm入参失败')
}
module.exports = args
