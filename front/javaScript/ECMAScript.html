let ​

  • 块级作用域：代码执行时遇到花括号，会创建一个块级作用域，花括号结束，销毁块级作用域
  • 使用let不会有变量提升，因此，不能在定义let变量之前使用它,底层实现上，let声明的变量实际上也会有提升，但是，提升后会将其放入到“暂时性死区”，如果访问的变量位于暂时性死区，则会报错：“Cannot access 'a' before initialization”。当代码运行到该变量的声明语句时，会将其从暂时性死区中移除。

使用const声明常量 ​

  • 常量不可变，是指声明的常量的内存空间不可变，并不保证内存空间中的地址指向的其他空间不可变。
  • 常量的命名
    1. 特殊的常量：该常量从字面意义上，一定是不可变的，比如圆周率、月地距地或其他一些绝不可能变化的配置。通常，该常量的名称全部使用大写，多个单词之间用下划线分割
    2. 普通的常量：使用和之前一样的命名即可
  • 在for循环中，循环变量不可以使用常量

class MySet {
  constructor(iterator = []) {
    // 是否可迭代对象
    if (typeof iterator[Symbol.iterator] !== "function") {
      throw new TypeError(iterator + " is not iterable");
    }
    this._data = [];
    for (let item of iterator) {
      this.add(item);
    }

  }

  get size() {
    return this._data.length;
  };

  add(value) {
    if (!this.has(value)) {
      this._data.push(value);
    }
  };

  has(value) {
    for (let item of this._data) {
      if (item === value) {
        return true;
      }
    }
    return false;
  };

  delete(value) {
    for (let index = 0; index < this._data.length; index++) {
      if(isEqual(this._data[index], value)) {
        this._data.splice(index, 1);
        return true;
      }
    }
    return false;
      
  };
  clear() {
    this._data = [];
  };
  toString() {
    return `[${this._data.join(",")}]`;
  }
  
  *[Symbol.iterator]() {
    for (let item of this._data) {
      yield item;
    }
  }
  forEach(callback) {
    for (let item of this._data) {
      callback(item, item, this);
    }
  }
  
  isEqual(value, other) {
    return Object.is(value, other);
  }

}

// 测试
// 创建 MySet 实例
const mySet = new MySet([1, 2, 3, 3, 4]); // 自动去重

// 基本操作
console.log(mySet.size); // 4
console.log(mySet.has(2)); // true
console.log(mySet.has(5)); // false

// 添加元素
mySet.add(5);
mySet.add(5); // 重复添加无效
console.log(mySet.toString()); // "[1,2,3,4,5]"

// 删除元素
mySet.delete(3);
console.log([...mySet]); // [1, 2, 4, 5] (使用迭代器转换为数组)

// 特殊值处理
const specialSet = new MySet([NaN, +0, -0]);
console.log(specialSet.has(NaN)); // true
console.log(specialSet.has(-0)); // true
console.log(specialSet.has(+0)); // true

// forEach 使用
mySet.forEach((value) => {
  console.log(value * 2); // 2, 4, 8, 10
});

// 清空集合
mySet.clear();
console.log(mySet.size); // 0

// 从字符串创建集合
const strSet = new MySet("hello");
console.log([...strSet].join("")); // "helo" (自动去重)

class MyMap {
  constructor(iterator = []) {
    // 是否可迭代对象
    if (!isIterable(iterator)) {
      throw new TypeError(iterator + " is not iterable");
    }
    this._data = [];
    for (let item of iterator) {
      if (!isIterable(item)) {
        throw new TypeError(item + " is not iterable");
      }
      const itemIterator = item[Symbol.iterator]();
      const key = itemIterator.next().value;
      const value = itemIterator.next().value;
      this.set(key, value);
    }
  };
  get size() {
    return this._data.length;
  }
  set(key, value) {
      const item = this._getItem(key);
      if(item) {
           item.value = value;
      }else {
        this._data.push({key, value});
      }
    
  };
  get(key) {
      return this._getItem(key)?.value;
    
  };
  has(key) {
    return !!this._getItem(key);
  };
  delete(key) {
    for(let index = 0; index < this._data.length; index++) {
      const item = this._data[index];
      if (this.isEqual(item.key, key)) {
        this._data.splice(index, 1);
        return true;
      }
    }
    return false;
  };
  clear() {
    this._data = [];
  };
  *[Symbol.iterator]() {
    for (const item of this._data) {
      yield [item.key, item.value];
    }
  }
  forEach(callback) {
    return this._data.forEach((item) => {
      callback(item.value, item.key, this);
    })
  };
  isEqual(value, other) {
     return  Object.is(value, other)
  };
  isIterable(iterator) {
    return typeof iterator[Symbol.iterator] === "function";
  }
  _getItem(key) {
    for (const item of this._data) {
      if (this.isEqual(item.key, key)) {
        return item;
      }
    }
    return undefined;
  }
}

// 测试
const myMap = new MyMap([['a', 1], ['b', 2]]);

// 测试基本操作
console.log(myMap.size); // 2
console.log(myMap.get('a')); // 1
console.log(myMap.has('b')); // true

// 测试迭代器
for (const [key, value] of myMap) {
  console.log(key, value); // 'a' 1, 'b' 2
}

// 测试 forEach
myMap.forEach((value, key) => {
  console.log(key, value); // 'a' 1, 'b' 2
});

for await ... of ​

for await ... of迭代异步可迭代对象，每次迭代会等待异步操作完成，然后再继续迭代。在es2018新增

场景： 1.读取文件 2.处理异步数据流（如从数据库或API分页获取数据）

js
// 场景：假设我们开发一个从服务器上分页获取用户数据的应用，每次只能获取一页（如10个用户）的数据，直到所有数据加载完成。


/**
 * 根据传入的页码数返回对应页码的用户数据
 * @param {*} page 页码数
 */
async function fetchUser(page) {
  // 假设每页10个用户
  const pageSize = 10;

  // 模拟服务器延迟
  await new Promise((resolve) => setTimeout(resolve, 1500));

  // 总共的用户数据
  const totalUsers = 45;

  // 计算当前页对应的起始值和结束值
  const start = (page - 1) * pageSize;
  const end = Math.min(start + pageSize, totalUsers);

  // 如果没有更多的用户，就返回空数组
  if (start >= totalUsers) return [];

  // 生成当前页的用户数据
  const users = Array.from({ length: end - start }, (_, i) => {
    return {
      id: start + i + 1,
      name: `用户${start + i + 1}`,
    };
  });

  return users;
}

// 异步生成器函数：分页获取所有的用户
async function* fetchAllUsers() {
  let page = 1; // 一开始从第一页开始
  let hasMore = true; // 是否还有更多的用户数据

  while (hasMore) {
    // 根据当前页码拿到对应的用户数据
    const users = await fetchUser(page);
    if (users.length === 0) {
      hasMore = false;
    } else {
      yield users;
      page++;
    }
  }
}

async function processAllUsers() {
  let allUsers = [];

  for await (let users of fetchAllUsers()) {
    console.log(`当前页的用户数据：`, users);
    allUsers = allUsers.concat(users);
  }

  console.log(`所用的用户数据：`, allUsers);
}
processAllUsers();
