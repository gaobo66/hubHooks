发布订阅模式 ​

发布订阅模式是一种常见的设计模式，在许多场景中都有应用。我们可能已经在使用中接触过发布订阅模式，比如使用 addEventListener 方法来监听 DOM 事件、Vue 的事件总线机制等。

功能：

  • 订阅者订阅事件，发布者发布事件，订阅者收到事件后执行回调函数。on订阅/监听

  • emit 发布/注册

  • once 只执行一次

  • off解除绑定

tsx
interface EventFace {
    on: (event: string, callback: Function) => void;
    emit: (event: string, ...args: Array<any>) => void;
    off: (event: string, fn: Function) => void;
    once: (event: string, fn: Function) => void;
}

class Dispatch implements EventFace {
    events: Map<string, Function[]>;
    // 使用set的好处避免注册事件重复
    // events: Record<string,Set<(...arg:any[])=>void >>;
    constructor() {
        this.events = new Map();
    }

    // 订阅事件
    on(eventName: string, callback: Function) {
        //-------set-------
        // (this.events[eventName]??=new Set()).add(callback)
        //-------set-------
        
        // 判断事件中心是否存在
       if (this.events.has(eventName)) {
           const callbackList = this.events.get(eventName);
           callbackList && callbackList.push(callback);
       }else{
           // 不存在则创建
           this.events.set(eventName, [callback]);
       }
    }

    // 发布事件
    emit(eventName: string, ...args:any[]) {
        //-------set-------
        // this.events[eventName]?.forEach(fn => fn(...args));
        //-------set-------
       const callbackList = this.events.get(eventName);
        callbackList && callbackList.forEach(fn => fn(...args));
    }
    
    // 解除绑定
    off(eventName: string, callback: Function) {
        //-------set-------
        // this.events[eventName]?.delete(callback);
        //-------set-------
        const callbackList = this.events.get(eventName);
        if (callbackList) {
            callbackList.splice(callbackList.indexOf(callback), 1);
        }
    }

    // 只执行一次
    once(eventName: string, callback: (...args:any[])=>void) {
        // 创建一个函数，触发on()注册到事件中心,完成以后执行off()回收
        const cb = (...arg:any[])=>{
            callback(...arg);
            this.off(eventName, cb);    
        }
        this.on(eventName, cb);
    }
    
}

// 使用
const o = new Dispatch();

// 订阅事件 'abc'，输出参数和数字 1
o.on('abc', (...arg: Array<any>) => {
    console.log(arg, 1);
});

// 一次性订阅事件 'abc'，输出参数和字符串 'once'，只会触发一次
o.once('abc', (...arg: Array<any>) => {
    console.log(arg, 'once');
});

// 发布事件 'abc'，输出参数 1、true 和字符串 '小满'
o.emit('abc', 1, true, '小满');

// 再次发布事件 'abc'，输出参数 2、true 和字符串 '小满'
o.emit('abc', 2, true, '小满');
